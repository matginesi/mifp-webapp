# MIFP — COMANDI
# Eseguire sempre dalla root del progetto.


# =============================================================================
# AVVIO — TRE MODALITÀ
# =============================================================================

# 1. Locale: Flask sul sistema host. Crea/ripara automaticamente .venv
# e installa le dipendenze mancanti, incluso Flask.
./mifp local

# Locale in background
./mifp local -d

# 2. Docker locale: sviluppo e dati in MIFPAPP/DATABASE
./mifp docker

# Docker locale senza ricostruire l'immagine
./mifp docker --no-build

# 3. Docker production: Gunicorn e volume persistente
./mifp production

# Production pubblica con Caddy e HTTPS automatico
./mifp production example.org

# Porta diversa
./mifp local --port 3000
./mifp docker --port 3000
./mifp production --port 3000


# =============================================================================
# AMMINISTRATORE
# =============================================================================

# Configura interattivamente username e password.
# La password deve contenere almeno 10 caratteri.
# Se Docker è già avviato, il container web viene ricreato automaticamente.
./mifp admin

# Imposta direttamente lo username e chiede la nuova password
./mifp admin --username matteo

# Genera una nuova password casuale per uno username scelto
./mifp admin --username matteo

# Mostra le credenziali solo quando sono state generate automaticamente


# =============================================================================
# SCRAPER
# =============================================================================

# Default:
#   threads remoti: 16
#   mirror locale: /run/media/matteo/ARCHDISK/srv/http/mifp.eu

# Scraping remoto con thread predefiniti
./mifp scrape remote

# Scraping remoto scegliendo i thread
./mifp scrape remote --threads 8
./mifp scrape remote --threads 32

# Scraping del mirror locale predefinito
./mifp scrape local

# Scraping locale con un percorso diverso
./mifp scrape local --local-root /altro/percorso/mifp.eu

# Locale + remoto, con numero di thread scelto
./mifp scrape all --threads 24

# Aggiorna prima il mirror locale, poi esegue lo scraping
./mifp scrape local --update-mirror
./mifp scrape all --threads 24 --update-mirror

# Mantiene gli output esistenti invece di pulirli prima
./mifp scrape remote --threads 16 --no-fresh

# Cambia directory di output
./mifp scrape remote --outputs /tmp/mifp-scrapers


# =============================================================================
# DATABASE E PIPELINE COMPLETA
# =============================================================================

# Costruisce/importa il database da SCRAPERS/OUTPUTS
./mifp database

# Ricrea il database
./mifp database --fresh

# Ricrea senza scaricare asset remoti
./mifp database --fresh --skip-downloads

# Scraper + database in sequenza
./mifp refresh remote --threads 16
./mifp refresh local
./mifp refresh all --threads 24

# IMPORT_DATA è legacy e non fa parte del codebase; eventuali copie locali esterne non vengono lette o modificate automaticamente.


# =============================================================================
# PULIZIA
# =============================================================================

# Anteprima del reset completo: non elimina nulla
./mifp clean all --dry-run

# Pulisce cache, virtualenv e report generati
./mifp clean cache

# Pulisce i log della webapp e del launcher
./mifp clean logs

# Pulisce soltanto SCRAPERS/OUTPUTS
./mifp clean outputs

# Pulisce database, asset, backup, export e workspace generati
./mifp clean data

# Arresta e rimuove gli stack Docker, senza eliminare il volume production
./mifp clean docker

# Elimina anche il volume persistente production
./mifp clean docker --volumes

# Reset completo dei file generati; conserva MIFPAPP/CORE/.env
./mifp clean all

# Reset completo non interattivo
./mifp clean all --yes

# Reset completo includendo Docker
./mifp clean all --docker --yes

# Reset totale: include .env, credenziali generate e volume Docker production
./mifp clean all --env --docker --volumes --yes

# Eventuali copie locali legacy di IMPORT_DATA non sono mai incluse nei target di pulizia.


# =============================================================================
# TEST
# =============================================================================

./mifp test
./mifp test webapp
./mifp test scraper
./mifp test database
# I test browser mostrano file, avanzamento e percentuale come le altre suite.
./mifp test browser
./mifp test all

# Argomenti aggiuntivi per pytest
./mifp test webapp -- -x -vv


# =============================================================================
# STATO, LOG E ARRESTO
# =============================================================================

./mifp status

./mifp logs local
./mifp logs docker
./mifp logs production

./mifp stop local
./mifp stop docker
./mifp stop production
./mifp stop all

./mifp doctor
./mifp setup


# =============================================================================
# GIT — RESET DEL REMOTO, WORKING TREE LOCALE INVARIATA
# =============================================================================

# Prima verifica di essere su main e aggiorna la conoscenza del remoto:
git branch --show-current
git fetch origin

# Backup completo della vecchia storia e branch locale di sicurezza:
git bundle create ../mifp-prima-del-reset.bundle --all
git branch "backup/prima-reset-$(date +%Y%m%d-%H%M%S)"

# Costruisce un nuovo commit root usando un indice temporaneo.
# .gitignore decide cosa entra: webapp, runtime, test webapp e CI/CD.
# SCRAPERS, DATABASE e tools restano fisicamente nella root; IMPORT_DATA è stata rimossa dal codebase.
OLD_REMOTE="$(git rev-parse refs/remotes/origin/main)"
TMP_INDEX="$(mktemp)"
rm -f "$TMP_INDEX"
GIT_INDEX_FILE="$TMP_INDEX" git add -A
GIT_INDEX_FILE="$TMP_INDEX" git status --short
TREE="$(GIT_INDEX_FILE="$TMP_INDEX" git write-tree)"
COMMIT="$(printf '%s
' 'Initial webapp-only repository' | git commit-tree "$TREE")"
rm -f "$TMP_INDEX"

# Aggiorna main e l'indice senza cancellare o riscrivere i file locali:
git update-ref refs/heads/main "$COMMIT"
git reset --mixed "$COMMIT"

# Controlli obbligatori:
git status --short
git ls-files | sort

# Sostituisce main remoto soltanto se non è cambiato dopo git fetch:
git push --force-with-lease="main:$OLD_REMOTE" origin main
